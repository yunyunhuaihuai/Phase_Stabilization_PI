cmake_minimum_required(VERSION 3.16)
project(PhaseStabilizationPI VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(USE_REAL_HARDWARE "Build the Red Pitaya interface; Linux SDK required" OFF)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
if(MSVC)
    add_compile_options(/W4 /utf-8)
else()
    add_compile_options(-Wall -Wextra -Wpedantic)
endif()
find_package(Threads REQUIRED)
add_library(control_core control/DiscretePI.cpp config/ConfigParser.cpp)
target_include_directories(control_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_library(data_logger utils/DataLogger.cpp)
target_link_libraries(data_logger PUBLIC control_core Threads::Threads)
target_compile_definitions(data_logger PRIVATE USE_REAL_HARDWARE=$<BOOL:${USE_REAL_HARDWARE}>)
add_library(sim_hal hal/SimHAL.cpp)
target_include_directories(sim_hal PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(sim_hal PUBLIC Threads::Threads)
add_library(serial_protocol comm/RingBuffer.cpp comm/VSMDParser.cpp)
target_include_directories(serial_protocol PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(PhaseStabilization main.cpp)
target_link_libraries(PhaseStabilization PRIVATE control_core data_logger sim_hal serial_protocol)
target_compile_definitions(PhaseStabilization PRIVATE USE_REAL_HARDWARE=$<BOOL:${USE_REAL_HARDWARE}>)
if(USE_REAL_HARDWARE)
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
        message(FATAL_ERROR "Real hardware requires Linux and the Red Pitaya SDK")
    endif()
    set(REDPITAYA_ROOT /opt/redpitaya CACHE PATH "Red Pitaya SDK root")
    find_path(RP_INCLUDE_DIR rp.h HINTS ${REDPITAYA_ROOT}/include)
    find_library(RP_LIBRARY rp HINTS ${REDPITAYA_ROOT}/lib)
    if(NOT RP_INCLUDE_DIR OR NOT RP_LIBRARY)
        message(FATAL_ERROR "The Red Pitaya SDK header and library are required")
    endif()
    add_library(redpitaya_hal hal/LinuxHAL.cpp)
    target_compile_definitions(redpitaya_hal PUBLIC USE_REAL_HARDWARE=1)
    target_include_directories(redpitaya_hal PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${RP_INCLUDE_DIR})
    target_link_libraries(redpitaya_hal PUBLIC ${RP_LIBRARY} Threads::Threads rt serial_protocol)
    target_link_libraries(PhaseStabilization PRIVATE redpitaya_hal)
endif()
include(CTest)
if(BUILD_TESTING)
    foreach(name pi_tests config_tests safety_path_tests runaway_guard_recovery_tests vsmd_parser_tests logger_shutdown_tests integration_tests)
        add_executable(${name} tests/${name}.cpp)
        target_include_directories(${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
        target_compile_definitions(${name} PRIVATE USE_REAL_HARDWARE=0)
        target_link_libraries(${name} PRIVATE control_core data_logger sim_hal serial_protocol)
        file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/test-output/${name})
        add_test(NAME ${name} COMMAND ${name} ${CMAKE_CURRENT_SOURCE_DIR}/config.txt)
        set_tests_properties(${name} PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/test-output/${name} TIMEOUT 45)
    endforeach()
    if(NOT USE_REAL_HARDWARE)
        file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/test-output/simulation)
        add_test(NAME simulation_startup_and_loop COMMAND PhaseStabilization ${CMAKE_CURRENT_SOURCE_DIR}/config.txt --simulation-seconds 1)
        set_tests_properties(simulation_startup_and_loop PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/test-output/simulation TIMEOUT 45)
    endif()
endif()
